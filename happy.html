import tkinter as tk import random # 全局列表：存储所有弹窗窗口，用于统一关闭 all_windows = [] # ========== 可自定义配置 ========== # 弹窗随机文案，可自行增删修改 tips = [ "学业有成", "好好爱自己", "好好吃饭", "保持好心情", "身体健康", "顺顺利利", "别熬夜", "天凉了多穿衣服", "天天开心", "万事胜意", "平安喜乐", "心想事成", "多喝水", "早点休息", "加油呀", "你超棒的", "烦恼全消", "好运连连", "按时吃饭", "少玩手机" ] # 弹窗随机背景色 colors = [ "pink", "lightblue", "lightgreen", "lemonchiffon", "hotpink", "skyblue", "palegreen", "lavender", "peachpuff", "lightcyan", "honeydew", "mistyrose" ] # 弹窗尺寸 WIN_WIDTH = 150 WIN_HEIGHT = 80 # 弹窗总数量 POPUP_COUNT = 200 # 弹窗弹出间隔（毫秒），数值越小弹出越快 POPUP_SPEED = 25 # ================================== def create_random_popup(root): """在屏幕随机位置创建一个弹窗""" win = tk.Toplevel(root) # 获取屏幕分辨率，确保弹窗不会超出屏幕边界 screen_w = root.winfo_screenwidth() screen_h = root.winfo_screenheight() # 随机生成左上角坐标 x = random.randint(0, screen_w - WIN_WIDTH) y = random.randint(0, screen_h - WIN_HEIGHT) # 设置窗口大小和位置 win.geometry(f"{WIN_WIDTH}x{WIN_HEIGHT}+{x}+{y}") win.title("小提示") # 窗口置顶，始终显示在最前方 win.attributes("-topmost", 1) # 取消下方注释可开启无边框效果（更美观，没有标题栏） # win.overrideredirect(True) # 随机选择文案和背景色 show_text = random.choice(tips) bg_color = random.choice(colors) # 创建文字标签 tk.Label( win, text=show_text, bg=bg_color, font=("SimHei", 12, "bold"), fg="#333333" ).pack(fill="both", expand=True) all_windows.append(win) def close_all_windows(event): """按下空格键时，销毁所有弹窗并退出程序""" for win in all_windows: try: win.destroy() except: pass root.quit() def popup_loop(root, count=0): if count >= POPUP_COUNT: return create_random_popup(root) root.after(POPUP_SPEED, lambda: popup_loop(root, count + 1)) if __name__ == "__main__": # 创建主窗口并隐藏（所有弹窗都依赖这个主窗口） root = tk.Tk() root.withdraw() # 绑定键盘事件：按下空格键触发关闭 root.bind("", close_all_windows) # 启动弹窗动画（不会卡顿界面） popup_loop(root, 0) # 进入窗口消息循环 root.mainloop()
